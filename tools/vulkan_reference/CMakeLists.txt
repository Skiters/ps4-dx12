cmake_minimum_required(VERSION 3.24)
project(shadps4_reference_completion LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Reference completion observer requires Windows stack capture")
endif()
set(SHADPS4_REFERENCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../shadps4-original-e0a46675" CACHE PATH "Pinned immutable original source")
add_library(guest-completion-layer SHARED completion_layer.cpp completion_layer.def)
target_compile_features(guest-completion-layer PRIVATE cxx_std_20)
target_include_directories(guest-completion-layer PRIVATE "${SHADPS4_REFERENCE_ROOT}/externals/vulkan-headers/include" "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
target_compile_options(guest-completion-layer PRIVATE /W4 /WX /Zi)
target_link_options(guest-completion-layer PRIVATE /DEBUG:FULL)
configure_file(completion-layer.json completion-layer.json COPYONLY)
